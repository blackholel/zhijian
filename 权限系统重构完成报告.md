## 🎨 使用示例

### 基础权限装饰器

```python
# 知识库权限
@data.get("/info")
@require_kb_permission(Permission.READ, "db_id")
async def get_database_info(db_id: str, current_user: User = Depends(get_required_user)):
    return await knowledge_base.get_database_info(db_id)

# 系统级权限
@data.post("/")
@require_system_permission(Permission.CREATE)
async def create_database(current_user: User = Depends(get_required_user)):
    return await knowledge_base.create_database(...)

# 对话权限
@chat.get("/sessions/{session_id}")
@require_chat_permission(Permission.READ, "session_id")
async def get_chat_session(session_id: str, current_user: User = Depends(get_required_user)):
    return await chat_service.get_session(session_id)
```

### 动态权限检查

```python
# 复杂场景的动态权限检查
async def complex_permission_check(user_id: str, meta: dict):
    db_id = meta.get("db_id")
    if db_id:
        engine = PermissionEngine.get_instance()
        resource = KnowledgeBaseResource(db_id)
        context = PermissionContext(
            user_id=user_id,
            resource=resource,
            permission=Permission.READ,
            request_metadata=meta,
            timestamp=datetime.now()
        )
        
        result = await engine.check_permission(context)
        if not result.allowed:
            raise HTTPException(403, f"Permission denied: {result.reason}")
```
